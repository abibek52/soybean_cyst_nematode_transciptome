(R)
suppressMessages({

## -----------------------------------------------------------------
## 0. PATHS & GLOBAL PARAMETERS
## -----------------------------------------------------------------
BASE_DIR <- "/path/to/file"
GFF_FILE <- file.path(BASE_DIR, "heterodera_glycines.PRJNA381081.WBPS19.annotations.gff3/heterodera_glycines.PRJNA381081.WBPS19.annotations.gff3")
META_CSV <- file.path(BASE_DIR, "Time-Treatment-all.csv")

OUT_DIR <- "."
dir.create(file.path(OUT_DIR, "results"), showWarnings = FALSE, recursive = TRUE)
dir.create(file.path(OUT_DIR, "figures"), showWarnings = FALSE, recursive = TRUE)

LFC_T  <- 2     # |log2FoldChange| threshold
PADJ_T <- 0.05  # BH-adjusted p-value threshold

## -----------------------------------------------------------------
## 1. EXTRACT TX2GENE FROM GFF3
## -----------------------------------------------------------------
cat("Building TxDb from GFF3...\n")
txdb <- makeTxDbFromGFF(GFF_FILE, format = "gff3")

k <- keys(txdb, keytype = "TXNAME")
tx2gene <- AnnotationDbi::select(txdb, keys = k, columns = "GENEID", keytype = "TXNAME")

## -----------------------------------------------------------------
## 2. LOAD METADATA & MAP SALMON DIRECTORIES
## -----------------------------------------------------------------
meta_raw <- read.csv(META_CSV, row.names = 1, check.names = FALSE)
meta <- as.data.frame(t(meta_raw))

meta$days      <- as.character(meta$days)
meta$replicate <- as.character(meta$replicate)

# Drop outlier sample ST16 if present
meta <- meta[!grepl("^ST16", rownames(meta)), ]

sample_dirs <- list.dirs(BASE_DIR, full.names = FALSE, recursive = FALSE)
sample_dirs <- grep("_count$", sample_dirs, value = TRUE)
sample_dirs <- sample_dirs[!grepl("^ST16", sample_dirs)]
raw_accessions <- gsub("_count$", "", sample_dirs)

files <- file.path(BASE_DIR, sample_dirs, "quant.sf")
names(files) <- raw_accessions
stopifnot(all(file.exists(files)))

common_samples <- intersect(names(files), rownames(meta))
files <- files[common_samples]
meta  <- meta[common_samples, ]

# Rename sample identities to experimental condition labels
new_names <- meta$Rename
names(files) <- new_names
rownames(meta) <- new_names
stopifnot(all(names(files) == rownames(meta)))

# Factors setup
meta$genotype  <- factor(meta$genotype,  levels = c("S", "R"))
meta$treatment <- factor(meta$treatment, levels = c("Cntrl", "SCN", "SCN+SBA"))
meta$days      <- factor(meta$days,      levels = c("5", "30"))
meta$group     <- factor(paste(meta$genotype,
                               gsub("\\+", "_", meta$treatment),
                               meta$days, sep = "_"))

## -----------------------------------------------------------------
## 3. IMPORT QUANTIFICATIONS VIA TXIMPORT
## -----------------------------------------------------------------
cat("Importing transcript quantifications to gene level...\n")
txi <- tximport(
  files           = files,
  type            = "salmon",
  tx2gene         = tx2gene,
  ignoreTxVersion = FALSE
)

## -----------------------------------------------------------------
## 4. DESeq2 MODEL 1: ~group (Pairwise Contrasts)
## -----------------------------------------------------------------
dds <- DESeqDataSetFromTximport(txi = txi, colData = meta, design = ~ group)
keep <- rowSums(counts(dds)) >= 10
dds  <- dds[keep, ]
dds  <- DESeq(dds)

run_contrast <- function(dds_obj, g1, g2, name) {
  res <- results(dds_obj, contrast = c("group", g1, g2), alpha = PADJ_T)
  res <- as.data.frame(res)
  res <- res[order(res$padj), ]
  sig <- res[!is.na(res$padj) & res$padj < PADJ_T & abs(res$log2FoldChange) > LFC_T, ]
  
  write.csv(res, file.path(OUT_DIR, sprintf("results/DEG_%s_full.csv", name)))
  write.csv(sig, file.path(OUT_DIR, sprintf("results/DEG_%s_sig.csv", name)))
  cat(sprintf("[Model 1] %-22s: %d sig DEGs\n", name, nrow(sig)))
  nrow(sig)
}

comparisons <- list(
  c("S_SCN_5",      "S_SCN_30",     "S_SCN_5vs30"),
  c("S_SCN_5",      "S_SCN_SBA_5",  "S_SCNvsSCN+SBA_5"),
  c("S_SCN_30",     "S_SCN_SBA_30", "S_SCNvsSCN+SBA_30"),
  c("S_SCN_SBA_5",  "S_SCN_SBA_30", "S_SCN+SBA_5vs30"),
  c("R_SCN_5",      "R_SCN_30",     "R_SCN_5vs30"),
  c("R_SCN_5",      "R_SCN_SBA_5",  "R_SCNvsSCN+SBA_5"),
  c("R_SCN_30",     "R_SCN_SBA_30", "R_SCNvsSCN+SBA_30"),
  c("R_SCN_SBA_5",  "R_SCN_SBA_30", "R_SCN+SBA_5vs30")
)

summary_tab <- do.call(rbind, lapply(comparisons, function(cc) {
  n <- run_contrast(dds, cc[1], cc[2], cc[3])
  data.frame(comparison = cc[3], group1 = cc[1], group2 = cc[2], n_sig_DEGs = n)
}))
write.csv(summary_tab, file.path(OUT_DIR, "results/summary_group_comparisons.csv"), row.names = FALSE)

## -----------------------------------------------------------------
## 5. DESeq2 MODEL 2: ~treatment * genotype (Active Treatments Only)
## -----------------------------------------------------------------
summary_list <- list()

for (day in c("5", "30")) {
  sub_meta  <- meta[meta$days == day, ]
  sub_files <- files[rownames(sub_meta)]
  
  sub_txi <- tximport(sub_files, type = "salmon", tx2gene = tx2gene, ignoreTxVersion = FALSE)
  
  sub_meta$treatment <- factor(sub_meta$treatment, levels = c("Cntrl", "SCN", "SCN+SBA"))
  sub_meta$genotype  <- factor(sub_meta$genotype,  levels = c("S", "R"))
  
  dds_d <- DESeqDataSetFromTximport(
    txi     = sub_txi,
    colData = sub_meta,
    design  = ~ treatment * genotype
  )
  
  keep_d <- rowSums(counts(dds_d)) >= 10
  dds_d  <- dds_d[keep_d, ]
  dds_d  <- DESeq(dds_d)
  
  coef_names  <- resultsNames(dds_d)
  int_scn     <- grep("treatment.*SCN.*genotype.*R", coef_names, value = TRUE)
  int_scn_sba <- grep("treatment.*SCN.*SBA.*genotype.*R", coef_names, value = TRUE)
  
  # Only evaluate active treatments: SCN and SCN+SBA (Cntrl excluded)
  day_contrasts <- list(
    list(name = sprintf("RvsS_SCN_%sdpi", day),     contrast = list(c("genotype_R_vs_S", int_scn)),     desc = "SCN"),
    list(name = sprintf("RvsS_SCN+SBA_%sdpi", day), contrast = list(c("genotype_R_vs_S", int_scn_sba)), desc = "SCN+SBA")
  )
  
  for (item in day_contrasts) {
    res <- results(dds_d, contrast = item$contrast, alpha = PADJ_T)
    res <- as.data.frame(res)
    res <- res[order(res$padj), ]
    sig <- res[!is.na(res$padj) & res$padj < PADJ_T & abs(res$log2FoldChange) > LFC_T, ]
    
    write.csv(res, file.path(OUT_DIR, sprintf("results/DEG_%s_full.csv", item$name)))
    write.csv(sig, file.path(OUT_DIR, sprintf("results/DEG_%s_sig.csv", item$name)))
    
    cat(sprintf("[Model 2] %-22s (%s %sdpi): %d sig DEGs\n", item$name, item$desc, day, nrow(sig)))
    
    summary_list[[item$name]] <- data.frame(
      comparison = item$name,
      day        = day,
      treatment  = item$desc,
      n_sig_DEGs = nrow(sig)
    )
  }
}

summary_genotype_df <- do.call(rbind, summary_list)
write.csv(summary_genotype_df, file.path(OUT_DIR, "results/summary_genotype_comparisons.csv"), row.names = FALSE)

col_map <- c("NS" = "grey70", "Down" = "#1f77b4", "Up" = "#d62728")

classify <- function(res) {
  res$sig <- "NS"
  res$sig[!is.na(res$padj) & res$padj < PADJ_T & res$log2FoldChange >  LFC_T] <- "Up"
  res$sig[!is.na(res$padj) & res$padj < PADJ_T & res$log2FoldChange < -LFC_T] <- "Down"
  res$sig <- factor(res$sig, levels = c("NS", "Down", "Up"))
  res
}

sig_set <- function(name) {
  f <- file.path(OUT_DIR, sprintf("results/DEG_%s_sig.csv", name))
  if (file.exists(f)) rownames(read.csv(f, row.names = 1)) else character(0)
}

export_gene_list <- function(gene_vector, filename, description) {
  if (length(gene_vector) == 0) {
    df <- data.frame(gene_id = character(0), category = character(0), stringsAsFactors = FALSE)
    cat(sprintf("[Notice] %-45s: 0 genes (empty set)\n", filename))
  } else {
    df <- data.frame(gene_id = gene_vector, category = rep(description, length(gene_vector)), stringsAsFactors = FALSE)
    cat(sprintf("[Exported] %-45s: %d genes\n", filename, length(gene_vector)))
  }
  write.csv(df, file.path(OUT_DIR, sprintf("results/%s", filename)), row.names = FALSE)
}

# Native pheatmap file output.
draw_heatmap <- function(genes, samples, title, fname, w = 8.5, h = 11.2, show_rownames = TRUE) {
  genes <- intersect(genes, rownames(vst_mat))
  if (length(genes) < 2) return(NULL)
  
  mat <- vst_mat[genes, samples, drop = FALSE]
  mat_z <- t(scale(t(mat)))
  
  out_file <- file.path(OUT_DIR, sprintf("figures/%s.png", fname))
  
  pheatmap(
    mat_z,
    color             = colorRampPalette(rev(brewer.pal(9, "RdBu")))(100),
    annotation_col    = meta_hm[samples, , drop = FALSE],
    annotation_colors = ann_colors,
    show_rownames     = show_rownames,
    fontsize_row      = 6.5,
    fontsize_col      = 8,
    treeheight_row    = 35,       # Keeps left dendrogram compact and within margins
    treeheight_col    = 30,       # Balances column dendrogram height
    main              = title,
    filename          = out_file, # Native direct-to-file graphics pipeline
    width             = w,        # Inches
    height            = h         # Inches
  )
  cat(sprintf("[Heatmap Saved] %s (%.1f x %.1f in)\n", basename(out_file), w, h))
}

6. Protein sequence extraction from fasta file, using significantly expressed gene names

(R)
library(Biostrings)

BASE_DIR <- "/path/to/file"
OUT_DIR  <- file.path(getwd(), "results")
dir.create(OUT_DIR, showWarnings = FALSE, recursive = TRUE)

# 1. Load protein FASTA
fasta_candidates <- list.files(
  BASE_DIR,
  pattern     = "protein.*(\\.(fa|faa|fasta)(\\.gz)?)?$",
  recursive   = TRUE,
  full.names  = TRUE,
  ignore.case = TRUE
)

if (length(fasta_candidates) == 0) {
  stop("Protein FASTA file could not be located under: ", BASE_DIR)
}

fasta_in <- fasta_candidates[1]
proteins <- readAAStringSet(fasta_in)
cat(sprintf("Loaded FASTA with %d protein sequences.\n", length(proteins)))

# 2. Prepare multiple cleaned versions of FASTA headers for matching
raw_headers     <- names(proteins)
first_word_ids  <- sub("^([^ \t|]+).*", "\\1", raw_headers)
gene_ids_no_ver <- sub("\\.[0-9]+$", "", first_word_ids)

# Extract gene=... or transcript=... if stored inside description lines
wb_gene_ids <- ifelse(grepl("gene=", raw_headers),
                      sub(".*gene=([^ \t;]+).*", "\\1", raw_headers),
                      NA)

# 3. Locate all significant DEG CSV files
csv_files <- list.files(OUT_DIR, pattern = "^DEG_.*_sig\\.csv$", full.names = TRUE)

if (length(csv_files) == 0) {
  # Search entire working directory if not located directly in OUT_DIR
  csv_files <- list.files(".", pattern = "^DEG_.*_sig\\.csv$", recursive = TRUE, full.names = TRUE)
}

cat(sprintf("Processing %d DEG CSV files...\n\n", length(csv_files)))

# 4. Extract sequences for each DEG CSV
for (csv_file in csv_files) {
  deg_df <- read.csv(csv_file, row.names = 1)
  deg_genes <- unique(rownames(deg_df))
  
  if (length(deg_genes) == 0) {
    cat(sprintf("[Skipped] %-35s (0 DEGs in file)\n", basename(csv_file)))
    next
  }
  
  # Check across all header variations
  matched_idx <- which(
    raw_headers %in% deg_genes |
      first_word_ids %in% deg_genes |
      gene_ids_no_ver %in% deg_genes |
      wb_gene_ids %in% deg_genes
  )
  
  comp_name <- sub("^DEG_(.*)_sig\\.csv$", "\\1", basename(csv_file))
  out_fasta <- file.path(OUT_DIR, sprintf("Proteins_%s.fasta", comp_name))
  
  if (length(matched_idx) > 0) {
    writeXStringSet(proteins[matched_idx], filepath = out_fasta, format = "fasta")
    cat(sprintf("[Extracted] %-35s -> %d sequences written to %s\n",
                basename(csv_file), length(matched_idx), basename(out_fasta)))
  } else {
    cat(sprintf("[Failed]    %-35s -> 0 / %d DEGs matched in FASTA\n",
                basename(csv_file), length(deg_genes)))
  }
}

# 5. List created files
created_fastas <- list.files(OUT_DIR, pattern = "\\.fasta$", full.names = TRUE)
cat(sprintf("\nSuccessfully created %d FASTA file(s) in: %s\n", length(created_fastas), OUT_DIR))
